    <script>
        // ==========================================
        // 3.0 邏輯: 累積計算 (Accumulation)
        // ==========================================
        const inputs30 = ['currentAge', 'retireAge', 'currentEPF', 'monthlySalary', 'contribRate', 'dividendRate'];
        const accCtx = document.getElementById('accumulationChart').getContext('2d');
        
        // 初始化圖表 3.0
        let accChart = new Chart(accCtx, {
            type: 'line',
            data: {
                labels: [],
                datasets: [
                    {
                        label: '總資產成長',
                        data: [],
                        borderColor: '#3B82F6', // Blue
                        backgroundColor: 'rgba(59, 130, 246, 0.1)',
                        fill: true,
                        tension: 0.4
                    },
                    {
                        label: '體面生活門檻 (600k)',
                        data: [],
                        borderColor: '#EF4444', // Red dashed
                        borderDash: [5, 5],
                        pointRadius: 0,
                        borderWidth: 1
                    }
                ]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: { legend: { display: true, labels: { color: '#94a3b8' } } },
                scales: {
                    y: { grid: { color: '#334155' }, ticks: { color: '#94a3b8' } },
                    x: { ticks: { color: '#94a3b8' } }
                }
            }
        });

        function calculateProjection() {
            // 獲取數值
            let age = parseInt(document.getElementById('currentAge').value);
            let retireAge = parseInt(document.getElementById('retireAge').value);
            let balance = parseFloat(document.getElementById('currentEPF').value);
            let salary = parseFloat(document.getElementById('monthlySalary').value);
            let rate = parseFloat(document.getElementById('contribRate').value) / 100;
            let dividend = parseFloat(document.getElementById('dividendRate').value) / 100;

            // 更新顯示文字
            document.getElementById('contribDisp').innerText = (rate * 100).toFixed(0) + '%';
            document.getElementById('dividendDisp').innerText = (dividend * 100).toFixed(1) + '%';

            let years = retireAge - age;
            if (years <= 0) return;

            let labels = [];
            let dataPoints = [];
            let benchmark = [];
            let totalContributed = balance;
            
            // 模擬每年增長 (簡化為年計算)
            for (let i = 0; i <= years; i++) {
                labels.push(age + i + '歲');
                dataPoints.push(balance);
                benchmark.push(600000); // 600k 基準線

                // 計算下一年的複利 + 繳納
                let yearlyContribution = salary * rate * 12;
                let interest = balance * dividend;
                balance = balance + yearlyContribution + interest;
                
                if (i > 0) totalContributed += yearlyContribution;
            }

            // 更新 UI
            let finalVal = dataPoints[dataPoints.length - 1];
            document.getElementById('finalAmount').innerText = 'RM ' + Math.floor(finalVal).toLocaleString();
            document.getElementById('totalInterest').innerText = '+ RM ' + Math.floor(finalVal - totalContributed).toLocaleString();

            // 更新圖表
            accChart.data.labels = labels;
            accChart.data.datasets[0].data = dataPoints;
            accChart.data.datasets[1].data = benchmark;
            accChart.update();
        }

        // ==========================================
        // 3.1 邏輯: 生存模擬 (Decumulation)
        // ==========================================
        const surCtx = document.getElementById('survivalChart').getContext('2d');
        let surChart = new Chart(surCtx, {
            type: 'doughnut',
            data: {
                labels: ['可支撐', '不足'],
                datasets: [{ data: [10, 20], backgroundColor: ['#F59E0B', '#334155'], borderWidth: 0, cutout: '85%' }]
            },
            options: { plugins: { legend: { display: false }, tooltip: { enabled: false } } }
        });

        function calculateSurvival() {
            const savings = parseFloat(document.getElementById('savingsInput').value) || 0;
            const expense = parseFloat(document.getElementById('expenseInput').value) || 0;
            document.getElementById('expenseRange').value = expense;

            if (expense === 0) return;

            const years = savings / (expense * 12);
            document.getElementById('yearsDisplay').innerText = years.toFixed(1);

            // 風險判斷
            const riskTitle = document.getElementById('riskTitle');
            let color = '#F59E0B'; // 黃
            let text = "風險偏高 (High Risk)";

            if (years < 10) { color = '#EF4444'; text = "極度危險 (Critical)"; } // 紅
            else if (years > 20) { color = '#10B981'; text = "相對安全 (Safe)"; } // 綠

            riskTitle.innerText = text;
            riskTitle.style.color = color;
            surChart.data.datasets[0].backgroundColor = [color, '#334155'];
            surChart.data.datasets[0].data = [Math.min(years, 30), 30 - Math.min(years, 30)];
            surChart.update();
        }

        // 連動功能：將 3.0 結果填入 3.1
        function syncToSurvival() {
            const finalAmountText = document.getElementById('finalAmount').innerText;
            const cleanAmount = parseFloat(finalAmountText.replace(/[^0-9.-]+/g,""));
            document.getElementById('savingsInput').value = cleanAmount;
            calculateSurvival();
            document.getElementById('survival-section').scrollIntoView({ behavior: 'smooth' });
        }

        // 綁定所有事件
        inputs30.forEach(id => document.getElementById(id).addEventListener('input', calculateProjection));
        ['savingsInput', 'expenseInput', 'expenseRange'].forEach(id => document.getElementById(id).addEventListener('input', calculateSurvival));

        // 初始執行
        calculateProjection();
        calculateSurvival();

    </script>
