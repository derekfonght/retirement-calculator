    <script>
        // 初始化變數
        const savingsInput = document.getElementById('savingsInput');
        const savingsRange = document.getElementById('savingsRange');
        const expenseInput = document.getElementById('expenseInput');
        const expenseRange = document.getElementById('expenseRange');
        
        const yearsDisplay = document.getElementById('yearsDisplay');
        const riskTitle = document.getElementById('riskTitle');
        const ctx = document.getElementById('resultChart').getContext('2d');


        // 初始化圖表 (Chart.js)
        let chart = new Chart(ctx, {
            type: 'doughnut',
            data: {
                labels: ['可支撐', '不足'],
                datasets: [{
                    data: [10.4, 20], // 預設值
                    backgroundColor: ['#F59E0B', '#334155'], // 黃色 / 深灰
                    borderWidth: 0,
                    cutout: '85%'
                }]
            },
            options: { pointerEvents: 'none', plugins: { legend: { display: false }, tooltip: { enabled: false } } }
        });


        // 核心計算邏輯
        function updateCalculator() {
            const savings = parseFloat(savingsInput.value) || 0;
            const expense = parseFloat(expenseInput.value) || 0;


            // 連動 Range Slider
            savingsRange.value = savings;
            expenseRange.value = expense;


            if (expense === 0) return;


            // 計算年限
            const years = savings / (expense * 12);
            const yearsFormatted = years.toFixed(1);


            // 更新數字顯示
            yearsDisplay.innerText = yearsFormatted;


            // 風險判斷邏輯
            let riskColor = '#F59E0B'; // 黃色
            let riskText = "風險偏高 (High Risk)";


            if (years < 5) {
                riskColor = '#EF4444'; // 紅色
                riskText = "極度危險 (Critical)";
            } else if (years >= 20) {
                riskColor = '#10B981'; // 綠色
                riskText = "相對安全 (Safe)";
            } else if (years >= 10) {
                riskColor = '#F59E0B'; // 黃色
                riskText = "風險偏高 (High Risk)";
            }


            // 更新 UI 顏色與文字
            riskTitle.innerText = riskText;
            riskTitle.className = `text-2xl font-bold mb-1`;
            riskTitle.style.color = riskColor;


            // 更新圖表
            chart.data.datasets[0].backgroundColor = [riskColor, '#334155'];
            // 這裡做一個視覺效果：如果超過30年就滿圈，否則按比例
            const maxYearsRef = 30; 
            const fillValue = Math.min(years, maxYearsRef);
            chart.data.datasets[0].data = [fillValue, maxYearsRef - fillValue];
            chart.update();
        }


        // 綁定事件監聽
        [savingsInput, savingsRange].forEach(el => {
            el.addEventListener('input', (e) => {
                savingsInput.value = e.target.value;
                updateCalculator();
            });
        });


        [expenseInput, expenseRange].forEach(el => {
            el.addEventListener('input', (e) => {
                expenseInput.value = e.target.value;
                updateCalculator();
            });
        });


        // 初始執行一次
        updateCalculator();
    </script>
